  function handleKeyDown(key) {
    if (!state.active) return false;
    if (state.modal) {
      if (key === "B") { closeDialog(state.modal); return true; }
      return true;
    }
    if (["UP", "DOWN", "LEFT", "RIGHT"].includes(key)) { setMove(key.toLowerCase(), true); return true; }
    if (key === "L") { openAvatarEditor(); return true; }
    if (key === "R") { openChat(); return true; }
    if (key === "SELECT") { openRoomMenu(); return true; }
    if (key === "B") { openRoomMenu(); return true; }
    return true;
  }

  function handleKeyUp(key) {
    if (!state.active) return false;
    if (["UP", "DOWN", "LEFT", "RIGHT"].includes(key)) setMove(key.toLowerCase(), false);
    return true;
  }

  async function init() {
    try {
      await window.ML3DAvatar.ready();
      state.profile = window.ML3DAvatar.loadProfile();
      injectUI();
      loadArena().catch(() => {});
    } catch (e) { console.error("ML3D Link init:", e); }
  }

  window.ml3dLink = {
    get active() { return state.active; },
    handleKeyDown,
    handleKeyUp,
    open: openSetup,
    state
  };

  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", init, { once: true });
  else init();
})();
