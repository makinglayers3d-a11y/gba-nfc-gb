  function openRoomMenu() {
    const hostBox = $("#ml3d-link-host-menu"), guestBox = $("#ml3d-link-guest-menu");
    hostBox.hidden = !state.host; guestBox.hidden = !state.guest;
    if (state.host) {
      $("#ml3d-link-room-name-input").value = state.host.room.name || "";
      $("#ml3d-link-room-game-input").value = state.host.room.game || "";
      $("#ml3d-link-room-max-input").value = String(state.host.room.maxPlayers || 2);
      renderRoomPlayers();
    }
    openDialog("ml3d-link-room-dialog");
  }

  function renderRoomPlayers() {
    const box = $("#ml3d-link-player-management"); if (!box) return;
    box.textContent = "";
    for (const p of state.players.values()) {
      const row = document.createElement("div"); row.className = "ml3d-link-manage-player";
      const label = document.createElement("span"); label.innerHTML = `<i class="ml3d-link-quality ${qualityClass(p.quality)}"></i>${escapeHtml(p.name)}${p.host ? " ★" : ""}`;
      row.append(label);
      if (state.host && !p.host) {
        const kick = document.createElement("button"); kick.type = "button"; kick.textContent = "EXPULSAR"; kick.addEventListener("click", () => kickPlayer(p.id)); row.append(kick);
      }
      box.append(row);
    }
  }

  function escapeHtml(text) {
    return String(text ?? "").replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#039;" }[c]));
  }

  function optionList(values) { return values.map((v) => `<option value="${v}">${v}</option>`).join(""); }
  function numericOptions(max) { return Array.from({ length: max }, (_, i) => `<option value="${i + 1}">${i + 1}</option>`).join(""); }
