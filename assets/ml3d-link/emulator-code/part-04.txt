(); setStatus("Sala cerrada");
  }

  async function leaveRoom(callApi = true) {
    const g = state.guest;
    if (g) {
      if (callApi) { try { await api(`/v1/rooms/${encodeURIComponent(g.room.id)}/joins/${encodeURIComponent(g.join.id)}/leave`, { method: "POST", token: g.token }); } catch {} }
      clearInterval(g.pollTimer); clearInterval(g.heartbeat); try { g.pc?.close(); } catch {}
    }
    state.guest = null; state.players.clear(); state.localId = null; deactivateLobby(); setStatus("Fuera de la sala");
  }

  async function saveRoomSettings(clearPassword = false) {
    if (!state.host) return;
    const body = {
      name: $("#ml3d-link-room-name-input").value,
      game: $("#ml3d-link-room-game-input").value,
      maxPlayers: Number($("#ml3d-link-room-max-input").value)
    };
    const pass = $("#ml3d-link-room-password-input").value;
    if (clearPassword) body.clearPassword = true; else if (pass) body.password = pass;
    try {
      const data = await api(`/v1/rooms/${encodeURIComponent(state.host.room.id)}/settings`, { method: "POST", token: state.host.token, body });
      state.host.room = { ...state.host.room, ...data.room }; updateRoomHeader(state.host.room);
      sendAll({ type: "lobby:room", room: state.host.room, time: Date.now() });
      openRoomMenu();
    } catch (e) { showError(e); }
  }

  async function kickPlayer(joinId) {
    if (!state.host) return;
    const peer = state.host.peers.get(joinId);
    try {
      await api(`/v1/rooms/${encodeURIComponent(state.host.room.id)}/joins/${encodeURIComponent(joinId)}/kick`, { method: "POST", token: state.host.token });
      safeSend(peer?.channel, { type: "lobby:kicked" });
      try { peer?.pc.close(); } catch {}
      state.host.peers.delete(joinId); state.players.delete(joinId); sendAll({ type: "lobby:player-left", playerId: joinId }); renderPlayers();
    } catch (e) { showError(e); }
  }

  }

  function sendChat(text) {
    text = cleanText(text, 90); if (!text || !state.localId) return;
    showBubble(state.localId, text);
    if (state.host) sendAll({ type: "lobby:chat", playerId: state.localId, text, time: Date.now() });
    else if (state.guest) safeSend(state.guest.channel, { type: "lobby:chat", text, time: Date.now() });
  }

  function moveTick() {
    if (!state.active || !state.localId || !state.held.size) return;
    const p = state.players.get(state.localId); if (!p) return;
    let dx = 0, dy = 0;
    if (state.held.has("left")) dx -= 1;
    if (state.held.has("right")) dx += 1;
    if (state.held.has("up")) dy -= 1;
    if (state.held.has("down")) dy += 1;
    if (!dx && !dy) return;
    const speed = 1.15;
    p.x = clamp(p.x + dx * speed, 8, 92, p.x);
    p.y = clamp(p.y + dy * speed, 29, 84, p.y);
    if (Math.abs(dx) > Math.abs(dy)) p.direction = dx < 0 ? "left" : "right";
    else p.direction = dy < 0 ? "up" : "down";
    state.lastDirection = p.direction;
    state.walkFrame = (state.walkFrame + 1) % 4;
    p.frame = [1, 2, 3, 2][state.walkFrame];
    renderPlayers(); sendMovement(p);
  }

  function ensureMoveTimer() { if (!state.moveTimer) state.moveTimer = setInterval(moveTick, MOVE_TICK); }
  function stopMoveTimer() { clearInterval(state.moveTimer); state.moveTimer = null; }

  function setMove(dir, down) {
    if (down) state.held.add(dir); else state.held.delete(dir);
    if (!down && !state.held.size && state.localId) {
      const p = state.players.get(state.localId); if (p) { p.frame = 0; renderPlayers(); sendMovement(p); }
    }
  }

    closeAllDialogs();
    const d = document.getElementById(id); if (!d) return;

  }

  function closeDialog(id) {
    const d = document.getElementById(id); if (!d) return;
    if (d.open && d.close) d.close(); else d.removeAttribute("open");
    if (state.modal === id) state.modal = null;
  }

  }
